01. Veri Tipleri (Primitive Types)

1. string

Metinsel veri tipidir.


                let name = "JavaScript";
                console.log(typeof name); // "string"

2. number

Sayısal veri tipidir (tam sayılar, ondalıklı sayılar, Infinity, NaN).


                let sayi = 42;
                console.log(typeof sayi); // "number"

3. boolean

Mantıksal değerlerdir (true veya false).


                let durum = true;
                console.log(typeof durum); // "boolean"

4. undefined vs null

undefined tanımlanmamış bir değişkeni temsil eder, null ise boş veya nesne olmayan bir değeri temsil eder.


                let tanimsiz;
                console.log(typeof tanimsiz); // "undefined"
        
                let bos = null;
                console.log(typeof bos); // "object" (hata)

5. symbol

Benzersiz ve değiştirilemez bir değerdir.


                let simge = Symbol('simge');
                console.log(typeof simge); // "symbol"

6. bigint

Büyük sayılar için kullanılan veri tipidir.


                let buyukSayi = 123456789012345678901234567890n;
                console.log(typeof buyukSayi); // "bigint"

7. typeof operatörü

Değerin veri tipini döndürür.


                console.log(typeof 42); // "number"
                console.log(typeof 'Merhaba'); // "string"
                console.log(typeof true); // "boolean"

8. type conversion

Veri tiplerini açıkça veya otomatik olarak dönüştürme.


                // Explicit conversion
                let strSayi = Number('123');
                console.log(strSayi); // 123
        
                let numStr = String(456);
                console.log(numStr); // "456"
        
                let boolNum = Boolean(1);
                console.log(boolNum); // true
        
                // Implicit conversion
                let toplam = '5' + 3;
                console.log(toplam); // "53"
        
                let toplam2 = '5' - 3;
                console.log(toplam2); // 2

02. Değişkenler (var, let, const)

1. var

var fonksiyon-scoped bir değişken tanımlar ve hoisting yapar.


                function example() {
                    console.log(x); // undefined (hoisting)
                    var x = 10;
                }
                example();

2. let

let blok-scoped bir değişken tanımlar ve Temporal Dead Zone (TDZ) vardır.


                if (true) {
                    console.log(y); // ReferenceError: Cannot access 'y' before initialization
                    let y = 20;
                }

3. const

const blok-scoped bir değişken tanımlar ve sabit bir değer alır, ancak nesne veya dizi gibi referans tipleri değiştirilebilir.


                const pi = 3.14;
                // pi = 3.15; // TypeError: Assignment to constant variable.
        
                const obj = { key: 'value' };
                obj.key = 'new value'; // Geçerli

4. Hoisting karşılaştırması

var hoist edilir, ancak let ve const hoist edilmez ancak TDZ içinde.


                console.log(z); // undefined
                var z = 30;
        
                // console.log(w); // ReferenceError: Cannot access 'w' before initialization
                let w = 40;

5. Global object (window/globalThis) ile ilişkisi

Global değişkenler, window veya globalThis nesnesi üzerinden erişilebilir.


                var globalVar = 'I am a global variable';
                console.log(window.globalVar); // "I am a global variable"
                console.log(globalThis.globalVar); // "I am a global variable"

03. Operatörler

1. Aritmetik Operatörler

Aritmetik operatörler sayısal değerleri işler.


                let a = 10;
                let b = 3;
                console.log(a + b); // 13
                console.log(a - b); // 7
                console.log(a * b); // 30
                console.log(a / b); // 3.3333333333333335
                console.log(a % b); // 1
                console.log(2 ** 3); // 8

2. Atama Operatörleri

Değişkenlere değer atar.


                let x = 5;
                x += 3; // x = x + 3
                console.log(x); // 8
        
                let y = 10;
                y -= 4; // y = y - 4
                console.log(y); // 6
        
                let z = 2;
                z *= 3; // z = z * 3
                console.log(z); // 6
        
                let w = 10;
                w /= 2; // w = w / 2
                console.log(w); // 5
        
                let v = 3;
                v **= 2; // v = v ** 2
                console.log(v); // 9

3. Karşılaştırma Operatörleri

Değerleri karşılaştırır.


                let num1 = 5;
                let num2 = '5';
                console.log(num1 == num2); // true (loose equality)
                console.log(num1 === num2); // false (strict equality)
        
                let str1 = 'hello';
                let str2 = 'world';
                console.log(str1 > str2); // true (lexicographical comparison)
        
                let a = 10;
                let b = 20;
                console.log(a < b); // true

4. Mantıksal Operatörler

Mantıksal değerleri işler.


                let p = true;
                let q = false;
        
                console.log(p && q); // false
                console.log(p || q); // true
                console.log(!p); // false
        
                let r;
                console.log(r ?? 'default'); // "default" (nullish coalescing)

5. Ternary Operatörü

Koşula göre farklı değerler döndürür.


                let age = 18;
                let message = (age >= 18) ? 'Adult' : 'Minor';
                console.log(message); // "Adult"

6. Optional Chaining

Nesnelerin özelliklerini güvenli bir şekilde erişir.


                const user = {
                    name: 'John',
                    address: {
                        city: 'New York'
                    }
                };
        
                console.log(user?.address?.city); // "New York"
                console.log(user?.address?.zip); // undefined

7. Spread Operatörü

Dizi veya nesneleri genişletir.


                let arr1 = [1, 2, 3];
                let arr2 = [...arr1, 4, 5];
                console.log(arr2); // [1, 2, 3, 4, 5]
        
                let obj1 = { a: 1, b: 2 };
                let obj2 = { ...obj1, c: 3 };
                console.log(obj2); // { a: 1, b: 2, c: 3 }

8. Rest Operatörü

Fonksiyon parametrelerini bir diziye dönüştürür.


                function sum(...numbers) {
                    return numbers.reduce((acc, num) => acc + num, 0);
                }
        
                console.log(sum(1, 2, 3, 4)); // 10

04. Kontrol Yapıları

1. if-else if-else (zincirleme örnek)

Koşullara göre farklı işlemler yapar.


                let sayi = 10;
                if (sayi > 0) {
                    console.log('Pozitif');
                } else if (sayi < 0) {
                    console.log('Negatif');
                } else {
                    console.log('Sıfır');
                }

2. switch-case (fallthrough, multiple cases)

Bir değere göre farklı işlemler yapar.


                let gun = 'Pazartesi';
                switch (gun) {
                    case 'Pazartesi':
                        console.log('Hafta başı');
                        break;
                    case 'Cuma':
                    case 'Cumartesi':
                        console.log('Hafta sonu');
                        break;
                    default:
                        console.log('Geçerli bir gün değil');
                }

3. for (klasik, break/continue)

Döngü kurar.


                for (let i = 0; i < 5; i++) {
                    if (i === 3) {
                        continue; // 3'ü atla
                    }
                    console.log(i); // 0, 1, 2, 4
                }
        
                for (let i = 0; i < 5; i++) {
                    if (i === 3) {
                        break; // 3'e kadar döngü
                    }
                    console.log(i); // 0, 1, 2
                }

4. for...of (array iteration)

Dizi üzerinde döngü kurar.


                let arr = [1, 2, 3, 4, 5];
                for (let item of arr) {
                    console.log(item); // 1, 2, 3, 4, 5
                }

5. for...in (object iteration, hasOwnProperty uyarısı)

Nesne üzerinde döngü kurar.


                let obj = { a: 1, b: 2, c: 3 };
                for (let key in obj) {
                    if (obj.hasOwnProperty(key)) {
                        console.log(key, obj[key]); // "a 1", "b 2", "c 3"
                    }
                }

6. while

Döngü kurar.


                let i = 0;
                while (i < 5) {
                    console.log(i); // 0, 1, 2, 3, 4
                    i++;
                }

7. do-while

Döngü kurar ve en az bir kez çalışır.


                let i = 0;
                do {
                    console.log(i); // 0, 1, 2, 3, 4
                    i++;
                } while (i < 5);

05. String Metotları

1. length

Dizinin uzunluğunu döndürür.


                let str = "Merhaba";
                console.log(str.length); // 7

2. toUpperCase() / toLowerCase()

Diziyi büyük veya küçük harf yapar.


                let str = "Merhaba";
                console.log(str.toUpperCase()); // "MERHABA"
                console.log(str.toLowerCase()); // "merhaba"

3. trim()

Dizinin başındaki ve sonundaki boşlukları kaldırır.


                let str = "   Merhaba   ";
                console.log(str.trim()); // "Merhaba"

4. slice() / substring()

Dizinin belirli bir kısmını döndürür.


                let str = "Merhaba";
                console.log(str.slice(1, 5)); // "erha"
                console.log(str.substring(1, 5)); // "erha"

5. split() / join()

Diziyi belirli bir karaktere göre böler veya dizi oluşturur.


                let str = "Merhaba,Dünya";
                console.log(str.split(",")); // ["Merhaba", "Dünya"]
        
                let arr = ["Merhaba", "Dünya"];
                console.log(arr.join(",")); // "Merhaba,Dünya"

6. replace() / replaceAll()

Diziyi belirli bir değeri başka bir değerle değiştirir.


                let str = "Merhaba Dünya";
                console.log(str.replace("Dünya", "JavaScript")); // "Merhaba JavaScript"
        
                let str2 = "Merhaba Dünya";
                console.log(str2.replaceAll("a", "o")); // "Merohbo Duwro"

7. includes() / startsWith() / endsWith()

Dizinin belirli bir değeri içerip içermediğini kontrol eder.


                let str = "Merhaba Dünya";
                console.log(str.includes("Dünya")); // true
                console.log(str.startsWith("Merhaba")); // true
                console.log(str.endsWith("Dünya")); // true

8. indexOf() / lastIndexOf()

Dizinin belirli bir değeri ilk veya son olarak hangi indeksde olduğunu döndürür.


                let str = "Merhaba Dünya";
                console.log(str.indexOf("Dünya")); // 7
                console.log(str.lastIndexOf("a")); // 10

9. padStart() / padEnd()

Dizinin belirli bir uzunlukta olması için başındaki veya sonundaki karakterleri ekler.


                let str = "123";
                console.log(str.padStart(5, '0')); // "00123"
                console.log(str.padEnd(5, '0')); // "12300"

10. repeat()

Diziyi belirli bir sayıda tekrar eder.


                let str = "Merhaba ";
                console.log(str.repeat(3)); // "Merhaba Merhaba Merhaba "

11. Template literals (``) - interpolation, multi-line, tagged templates

String ifadeleri daha esnek ve okunabilir hale getirir.


                let name = "John";
                let age = 30;
                console.log(`My name is ${name} and I am ${age} years old.`); // "My name is John and I am 30 years old."
        
                let str = `
                    Bu
                    çoklu
                    satırlı
                    bir
                    stringdir.
                `;
                console.log(str);
        
                function tag(strings, ...values) {
                    return strings.reduce((acc, str, i) => acc + str + (i < values.length ? values[i] : ''), '');
                }
        
                let message = tag`Hello, ${name}! You are ${age} years old.`;
                console.log(message); // "Hello, John! You are 30 years old."

06. Arrays (Diziler)

1. Array oluşturma

Dizileri oluşturmak için farklı yöntemler kullanabilirsiniz.


                let arr1 = [1, 2, 3];
                let arr2 = new Array(4, 5, 6);
                let arr3 = Array.of(7, 8, 9);
                let arr4 = Array.from('abc');

2. CRUD

Dizileri ekleme, silme ve güncelleme işlemleri.


                let arr = [1, 2, 3];
                arr.push(4); // [1, 2, 3, 4]
                console.log(arr);
        
                arr.pop(); // [1, 2, 3]
                console.log(arr);
        
                arr.unshift(0); // [0, 1, 2, 3]
                console.log(arr);
        
                arr.shift(); // [1, 2, 3]
                console.log(arr);

3. splice (ekleme/silme/güncelleme), slice (kopyalama)

Dizileri belirli bir konumda ekleme, silme veya güncelleme yapar.


                let arr = [1, 2, 3];
                arr.splice(1, 0, 'a', 'b'); // [1, 'a', 'b', 2, 3]
                console.log(arr);
        
                let arrCopy = arr.slice(1, 4); // ['a', 'b', 2]
                console.log(arrCopy);

4. concat (birleştirme)

Dizileri birleştirir.


                let arr1 = [1, 2];
                let arr2 = [3, 4];
                let mergedArr = arr1.concat(arr2); // [1, 2, 3, 4]
                console.log(mergedArr);

5. Iteration

Dizileri döngü ile işlemek için farklı metotlar kullanabilirsiniz.


                let arr = [1, 2, 3];
        
                // forEach
                arr.forEach(item => console.log(item)); // 1, 2, 3
        
                // map
                let doubled = arr.map(item => item * 2); // [2, 4, 6]
                console.log(doubled);
        
                // filter
                let even = arr.filter(item => item % 2 === 0); // [2, 4]
                console.log(even);
        
                // reduce
                let sum = arr.reduce((acc, item) => acc + item, 0); // 6
                console.log(sum);
        
                // reduceRight
                let sumRight = arr.reduceRight((acc, item) => acc + item, 0); // 6
                console.log(sumRight);

6. find, findIndex, some, every

Dizileri belirli bir koşula göre arama ve kontrol yapar.


                let arr = [1, 2, 3];
        
                // find
                let found = arr.find(item => item === 2); // 2
                console.log(found);
        
                // findIndex
                let foundIndex = arr.findIndex(item => item === 2); // 1
                console.log(foundIndex);
        
                // some
                let hasEven = arr.some(item => item % 2 === 0); // true
                console.log(hasEven);
        
                // every
                let allPositive = arr.every(item => item > 0); // true
                console.log(allPositive);

7. flat, flatMap

Dizileri iç içe yapıları düzleştirmek veya dönüştürmek için kullanılır.


                let arr = [1, 2, [3, 4]];
                let flatArr = arr.flat(); // [1, 2, 3, 4]
                console.log(flatArr);
        
                let arr2 = [1, 2, 3];
                let flatMapArr = arr2.flatMap(item => [item * 2]); // [2, 4, 6]
                console.log(flatMapArr);

8. sort, reverse

Dizileri sıralamak veya tersine çevirmek için kullanılır.


                let arr = [3, 1, 2];
                arr.sort((a, b) => a - b); // [1, 2, 3]
                console.log(arr);
        
                arr.reverse(); // [3, 2, 1]
                console.log(arr);

9. includes, indexOf

Dizileri belirli bir değeri içerip içermediğini kontrol etmek veya indeksini bulmak için kullanılır.


                let arr = [1, 2, 3];
                console.log(arr.includes(2)); // true
                console.log(arr.indexOf(4)); // -1

10. Array destructuring

Dizileri değişkenlere atarak kullanmak için kullanılır.


                let arr = [1, 2, 3];
                let [a, b, c] = arr;
                console.log(a, b, c); // 1, 2, 3

11. Spread ile kopyalama ve birleştirme

Dizileri kopyalamak veya birleştirmek için spread operatörü kullanılır.


                let arr1 = [1, 2, 3];
                let arr2 = [...arr1]; // [1, 2, 3]
                console.log(arr2);
        
                let arr3 = [...arr1, 4, 5]; // [1, 2, 3, 4, 5]
                console.log(arr3);

07. Objects (Nesneler)

1. Object literal ve constructor

Nesneleri oluşturmak için farklı yöntemler kullanabilirsiniz.


                // Object literal
                let obj1 = {
                    name: "John",
                    age: 30
                };
        
                // Object constructor
                function Person(name, age) {
                    this.name = name;
                    this.age = age;
                }
                let obj2 = new Person("Jane", 25);

2. Property access (dot, bracket, optional chaining)

Nesnelerin özelliklerine erişmek için farklı yöntemler kullanabilirsiniz.


                let obj = {
                    name: "John",
                    age: 30
                };
        
                // Dot notation
                console.log(obj.name); // "John"
        
                // Bracket notation
                console.log(obj['age']); // 30
        
                // Optional chaining
                let obj2 = null;
                console.log(obj2?.name); // undefined

3. Shorthand property, computed property names

Nesnelerin özelliklerini daha kolay ve dinamik olarak oluşturmak için kullanılır.


                let name = "John";
                let age = 30;
        
                // Shorthand property
                let obj1 = { name, age };
        
                // Computed property names
                let key = "name";
                let obj2 = { [key]: "John" };

4. Object.keys(), Object.values(), Object.entries()

Nesnelerin özelliklerini ve değerlerini dizi olarak almak için kullanılır.


                let obj = {
                    name: "John",
                    age: 30
                };
        
                // Object.keys()
                console.log(Object.keys(obj)); // ["name", "age"]
        
                // Object.values()
                console.log(Object.values(obj)); // ["John", 30]
        
                // Object.entries()
                console.log(Object.entries(obj)); // [["name", "John"], ["age", 30]]

5. Object.assign() (shallow copy)

Nesneleri kopyalamak için kullanılır.


                let obj1 = { a: 1, b: 2 };
                let obj2 = Object.assign({}, obj1);
                console.log(obj2); // { a: 1, b: 2 }

6. Object.freeze(), Object.seal()

Nesneleri sabit veya kilitlemek için kullanılır.


                let obj = { a: 1, b: 2 };
        
                // Object.freeze()
                Object.freeze(obj);
                obj.a = 3; // No effect
                console.log(obj); // { a: 1, b: 2 }
        
                // Object.seal()
                obj.b = 4; // Allowed
                obj.c = 5; // Not allowed
                console.log(obj); // { a: 1, b: 4 }

7. Object destructuring (default values, alias, nested)

Nesneleri değişkenlere atarak kullanmak için kullanılır.


                let obj = {
                    name: "John",
                    age: 30,
                    address: {
                        city: "New York"
                    }
                };
        
                // Default values
                let { name, age = 25 } = obj;
                console.log(name, age); // "John" 30
        
                // Alias
                let { name: userName } = obj;
                console.log(userName); // "John"
        
                // Nested
                let { address: { city } } = obj;
                console.log(city); // "New York"

8. Spread ile kopyalama ve birleştirme

Nesneleri kopyalamak veya birleştirmek için spread operatörü kullanılır.


                let obj1 = { a: 1, b: 2 };
                let obj2 = { ...obj1 }; // { a: 1, b: 2 }
                console.log(obj2);
        
                let obj3 = { ...obj1, c: 3 }; // { a: 1, b: 2, c: 3 }
                console.log(obj3);

9. Getters ve Setters

Nesnelerin özelliklerine erişim ve ayarlama işlemlerini kontrol etmek için kullanılır.


                let obj = {
                    _name: "John",
                    get name() {
                        return this._name;
                    },
                    set name(value) {
                        if (value.length > 3) {
                            this._name = value;
                        } else {
                            console.log("Name must be at least 4 characters long.");
                        }
                    }
                };
        
                obj.name = "Jane"; // "Name must be at least 4 characters long."
                console.log(obj.name); // "John"

10. Optional chaining (?.) ile güvenli erişim

Nesnelerin özelliklerine güvenli bir şekilde erişmek için kullanılır.


                let obj = null;
                console.log(obj?.name); // undefined

08. Functions (Fonksiyonlar)

1. Function declaration vs expression vs arrow function

Fonksiyonları farklı şekillerde tanımlayabilirsiniz.


                // Function declaration
                function add(a, b) {
                    return a + b;
                }
        
                // Function expression
                let multiply = function(a, b) {
                    return a * b;
                };
        
                // Arrow function
                let subtract = (a, b) => a - b;

2. Default parameters

Fonksiyonlara varsayılan parametreler ekleyebilirsiniz.


                function greet(name = "Guest") {
                    return `Hello, ${name}!`;
                }
        
                console.log(greet()); // "Hello, Guest!"
                console.log(greet("Alice")); // "Hello, Alice!"

3. Rest parameters (...args)

Fonksiyona değişken sayıda parametre geçebilirsiniz.


                function sum(...numbers) {
                    return numbers.reduce((acc, num) => acc + num, 0);
                }
        
                console.log(sum(1, 2, 3)); // 6
                console.log(sum(4, 5, 6, 7)); // 22

4. Return values (early return)

Fonksiyonlarda erken döndürme yapabilirsiniz.


                function checkAge(age) {
                    if (age < 18) {
                        return "Too young";
                    }
                    return "Welcome!";
                }
        
                console.log(checkAge(15)); // "Too young"
                console.log(checkAge(20)); // "Welcome!"

5. IIFE (Immediately Invoked Function Expression)

Anında çalıştırılan fonksiyonlar oluşturabilirsiniz.


                (function() {
                    console.log("This is an IIFE!");
                })();

6. Higher-order functions (function return function)

Fonksiyonlar diğer fonksiyonları döndürebilir.


                function createMultiplier(factor) {
                    return function(number) {
                        return number * factor;
                    };
                }
        
                let double = createMultiplier(2);
                console.log(double(5)); // 10

7. Callback functions (setTimeout, forEach)

Fonksiyonları geri çağırmak için kullanabilirsiniz.


                setTimeout(() => {
                    console.log("This runs after 1 second");
                }, 1000);
        
                let numbers = [1, 2, 3];
                numbers.forEach(number => {
                    console.log(number);
                });

8. call(), apply(), bind() (explicit this binding)

                        zamanlayici = setTimeout(() => {
                            fonksiyon.apply(this, args);
                        }, gecikme);
                    };
                }
                // Kullanımı: Sunucuya her harfte değil, yazma bitince istek atar
                const guvenliArama = debounce(() => console.log("API İsteği atıldı"), 500);

2. Throttle Fonksiyonu

Yoğun event akışlarında, fonksiyonun sürekli tetiklenmesini engelleyerek her X milisaniyede **en fazla bir kez** çalışmasına izin verir. Akışı düzenli aralıklara böler.


                function throttle(fn, limit) {
                    let beklemede = false;
                    return function(...args) {
                        if (!beklemede) {
                            fn.apply(this, args);
                            beklemede = true;
                            setTimeout(() => beklemede = false, limit);
                        }
                    };
                }
                // Kullanımı: Sayfa kaydırılırken her milisaniyede değil, 200ms'de bir çalışır
                window.addEventListener('scroll', throttle(() => console.log("Kaydırılıyor"), 200));
